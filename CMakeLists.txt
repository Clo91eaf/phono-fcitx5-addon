cmake_minimum_required(VERSION 3.22)
project(phono-fcitx5-addon VERSION 0.1.3 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This addon currently targets Linux Fcitx5")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
include(GNUInstallDirs)
# Shared filesystem directories must be 0755 even with a permissive umask.
set(CMAKE_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS
    OWNER_READ OWNER_WRITE OWNER_EXECUTE
    GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
find_package(Threads REQUIRED)
find_package(Fcitx5Core 5.1.7 REQUIRED)
find_package(LibIMEPinyin 1.1.5 REQUIRED)
find_package(nlohmann_json 3.2 REQUIRED)
set(PHONO_SDK_PREFIX "" CACHE PATH "Installed phono-core SDK prefix (optional with CMAKE_PREFIX_PATH)")
if(PHONO_SDK_PREFIX)
    list(PREPEND CMAKE_PREFIX_PATH "${PHONO_SDK_PREFIX}")
endif()
if(PHONO_CORE_INCLUDE_DIR AND PHONO_CORE_LIBRARY)
    # An explicit local SDK/library override remains useful for development.
    add_library(phono::core SHARED IMPORTED)
    set_target_properties(phono::core PROPERTIES IMPORTED_LOCATION "${PHONO_CORE_LIBRARY}"
        INTERFACE_INCLUDE_DIRECTORIES "${PHONO_CORE_INCLUDE_DIR}")
else()
    find_package(phono-core CONFIG REQUIRED)
    get_target_property(PHONO_CORE_LIBRARY phono::core IMPORTED_LOCATION_RELEASE)
    if(NOT PHONO_CORE_LIBRARY)
        get_target_property(PHONO_CORE_LIBRARY phono::core IMPORTED_LOCATION_RELWITHDEBINFO)
    endif()
    if(NOT PHONO_CORE_LIBRARY)
        get_target_property(PHONO_CORE_LIBRARY phono::core IMPORTED_LOCATION)
    endif()
    get_filename_component(_phono_sdk_libdir "${PHONO_CORE_LIBRARY}" DIRECTORY)
    get_filename_component(_phono_sdk_prefix "${_phono_sdk_libdir}" DIRECTORY)
    set(PHONO_CORE_LICENSE_ROOT "${_phono_sdk_prefix}/share/licenses/phono-core" CACHE PATH
        "Core SDK redistribution notices")
endif()
if(NOT PHONO_CORE_LIBRARY MATCHES "\\.so(\\.[0-9.]+)?$")
    message(FATAL_ERROR "This build uses shared libphono_core.so. A static integration requires a separate build path with PHONO_DTYPE_SELECTIVE_BUILD=OFF")
endif()
find_file(PHONO_PINYIN_DICT sc.dict HINTS "${LIBIME_INSTALL_PKGDATADIR}" /usr/share/libime REQUIRED)
file(RELATIVE_PATH _phono_addon_dir "${FCITX_INSTALL_PREFIX}" "${FCITX_INSTALL_ADDONDIR}")
file(RELATIVE_PATH _phono_data_dir "${FCITX_INSTALL_PREFIX}" "${FCITX_INSTALL_PKGDATADIR}")
set(PHONO_FCITX_ADDON_DIR "${_phono_addon_dir}" CACHE STRING "Fcitx addon path relative to install prefix")
set(PHONO_FCITX_DATA_DIR "${_phono_data_dir}" CACHE STRING "Fcitx data path relative to install prefix")
foreach(_phono_dir PHONO_FCITX_ADDON_DIR PHONO_FCITX_DATA_DIR)
    if(IS_ABSOLUTE "${${_phono_dir}}" OR "${${_phono_dir}}" MATCHES "(^|/)\\.\\.(/|$)")
        message(FATAL_ERROR "${_phono_dir} must stay relative to CMAKE_INSTALL_PREFIX")
    endif()
endforeach()
add_library(phono_runtime STATIC src/worker.cpp src/core.cpp src/presentation.cpp)
target_include_directories(phono_runtime PUBLIC src)
target_link_libraries(phono_runtime PUBLIC Threads::Threads nlohmann_json::nlohmann_json phono::core)
add_library(phono_engine STATIC src/addon.cpp src/shuangpin.cpp)
target_link_libraries(phono_engine PUBLIC phono_runtime Fcitx5::Core LibIME::Pinyin)
include(cmake/I18n.cmake)
target_link_libraries(phono_engine PUBLIC phono_i18n)
target_compile_definitions(phono_engine PRIVATE PHONO_PINYIN_DICT="${PHONO_PINYIN_DICT}")
add_library(phono MODULE src/module.cpp)
target_link_libraries(phono PRIVATE phono_engine)
add_dependencies(phono phono_translations)
option(PHONO_BUNDLE_CORE "Install libphono_core beside the addon in a private directory" ON)
if(PHONO_BUNDLE_CORE)
    set_target_properties(phono PROPERTIES INSTALL_RPATH "$ORIGIN/phono")
    install(FILES "${PHONO_CORE_LIBRARY}" DESTINATION "${PHONO_FCITX_ADDON_DIR}/phono")
endif()
target_compile_options(phono_engine PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(phono_runtime PRIVATE -Wall -Wextra -Wpedantic)
configure_file(data/phono-addon.conf.in data/addon/phono.conf @ONLY)
configure_file(data/phono-inputmethod.conf data/inputmethod/phono.conf COPYONLY)
configure_file(data/phono-shuangpin.conf data/inputmethod/phono-shuangpin.conf COPYONLY)
install(TARGETS phono LIBRARY DESTINATION "${PHONO_FCITX_ADDON_DIR}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/data/addon/phono.conf" DESTINATION "${PHONO_FCITX_DATA_DIR}/addon")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/data/inputmethod/phono.conf"
    "${CMAKE_CURRENT_BINARY_DIR}/data/inputmethod/phono-shuangpin.conf" DESTINATION "${PHONO_FCITX_DATA_DIR}/inputmethod")
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME package_tests COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/package_tests.py")
    add_test(NAME sdk_tests COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/sdk_tests.py")
    add_executable(worker_tests tests/worker_tests.cpp)
    target_link_libraries(worker_tests PRIVATE phono_runtime)
    add_test(NAME worker_tests COMMAND worker_tests)
    add_executable(addon_tests tests/addon_tests.cpp)
    target_link_libraries(addon_tests PRIVATE phono_engine)
    add_test(NAME addon_tests COMMAND addon_tests)
    add_executable(memory_tests tests/memory_tests.cpp)
    target_link_libraries(memory_tests PRIVATE phono_engine)
    add_test(NAME memory_tests COMMAND memory_tests)
    set_tests_properties(memory_tests PROPERTIES TIMEOUT 30 ENVIRONMENT
        "XDG_CONFIG_HOME=${CMAKE_CURRENT_BINARY_DIR}/memory-env/config;XDG_DATA_HOME=${CMAKE_CURRENT_BINARY_DIR}/memory-env/data")
    add_executable(load_tests tests/load_tests.cpp)
    target_include_directories(load_tests PRIVATE src)
    target_link_libraries(load_tests PRIVATE Fcitx5::Core LibIME::Pinyin nlohmann_json::nlohmann_json)
    add_dependencies(load_tests phono)
    add_test(NAME load_tests COMMAND load_tests)
    add_executable(shuangpin_tests tests/shuangpin_tests.cpp src/shuangpin.cpp)
    target_include_directories(shuangpin_tests PRIVATE src)
    target_link_libraries(shuangpin_tests PRIVATE LibIME::Pinyin)
    add_test(NAME shuangpin_tests COMMAND shuangpin_tests)
    add_executable(presentation_tests tests/presentation_tests.cpp)
    target_link_libraries(presentation_tests PRIVATE phono_engine)
    add_test(NAME presentation_tests COMMAND presentation_tests)
    add_executable(fdstream_tests tests/fdstream_tests.cpp)
    target_include_directories(fdstream_tests PRIVATE src)
    target_link_libraries(fdstream_tests PRIVATE Fcitx5::Utils)
    add_test(NAME fdstream_tests COMMAND fdstream_tests)
    add_executable(feature_tests tests/feature_tests.cpp)
    target_include_directories(feature_tests PRIVATE src)
    target_link_libraries(feature_tests PRIVATE Fcitx5::Core LibIME::Pinyin nlohmann_json::nlohmann_json phono_i18n)
    add_dependencies(feature_tests phono)
    add_test(NAME feature_tests COMMAND feature_tests)
    set_tests_properties(addon_tests PROPERTIES ENVIRONMENT
        "XDG_CONFIG_HOME=${CMAKE_CURRENT_BINARY_DIR}/test-env/config;XDG_DATA_HOME=${CMAKE_CURRENT_BINARY_DIR}/test-env/data")
    set_tests_properties(load_tests PROPERTIES ENVIRONMENT
        "XDG_CONFIG_HOME=${CMAKE_CURRENT_BINARY_DIR}/load-env/config;XDG_DATA_HOME=${CMAKE_CURRENT_BINARY_DIR}/load-env/data;FCITX_DATA_DIRS=${CMAKE_CURRENT_BINARY_DIR}/data:/usr/share/fcitx5;FCITX_ADDON_DIRS=${CMAKE_CURRENT_BINARY_DIR}")
    set_tests_properties(feature_tests PROPERTIES ENVIRONMENT
        "XDG_CONFIG_HOME=${CMAKE_CURRENT_BINARY_DIR}/feature-env/config;XDG_DATA_HOME=${CMAKE_CURRENT_BINARY_DIR}/feature-env/data;FCITX_DATA_DIRS=${CMAKE_CURRENT_BINARY_DIR}/data:${FCITX_INSTALL_PKGDATADIR};FCITX_ADDON_DIRS=${CMAKE_CURRENT_BINARY_DIR}:${FCITX_INSTALL_ADDONDIR}")
    set_tests_properties(package_tests worker_tests addon_tests load_tests shuangpin_tests presentation_tests feature_tests PROPERTIES TIMEOUT 30)
endif()
include(cmake/Packaging.cmake)

configure_file(ci/PKGBUILD.in ci/PKGBUILD @ONLY)
